import nltk
from nltk.chat.util import Chat, reflections

# Predefined question-answer pairs
pairs = [
    [r"hi|hello|hey",
     ["Hello! How can I help you today?", "Hi there!"]],

    [r"what is your name?",
     ["I am an AI chatbot created using Python and NLP."]],

    [r"how are you?",
     ["I'm doing great! How about you?"]],

    [r"what can you do?",
     ["I can answer simple questions and chat with you."]],

    [r"(.*) your creator?",
     ["I was created by a Python developer."]],

    [r"bye|goodbye",
     ["Goodbye! Have a great day."]]
]

# Create chatbot
chatbot = Chat(pairs, reflections)

print("AI Chatbot (type 'bye' to exit)")
chatbot.converse()
